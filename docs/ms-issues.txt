#!/usr/bin/env bash
# 注意：本脚本为 bash 语法，Windows 下请在 Git Bash / WSL 中运行（PowerShell 无法直接执行）
set -e

REPO="your-name/design-studio-site"   # ← 改成你的仓库

# ---- 1. 建 labels ----
declare -A LABELS=(
  ["front"]="0075ca:前端 / UI"
  ["backend"]="a2eeef:后端 / Strapi 接口"
  ["strapi"]="fbca04:Strapi 配置 / 内容模型"
  ["deploy"]="c5def5:部署 / DevOps"
  ["i18n"]="5319e7:中英双语"
  ["responsive"]="cfd3d7:响应式适配"
  ["seo"]="ededed:SEO / Meta"
  ["form"]="d4c5f9:联系表单"
  ["bug"]="d73a4a:Bug"
  ["enhancement"]="a2eeef:增强 / 可选"
  ["milestone:m1"]="c50e5a:M1 脚手架"
  ["milestone:m2"]="c50e5a:M2 mock 页面"
  ["milestone:m3"]="c50e5a:M3 Strapi 后台"
  ["milestone:m4"]="c50e5a:M4 联调"
  ["milestone:m5"]="c50e5a:M5 部署"
  ["milestone:m6"]="c50e5a:M6 验收"
)
for name in "${!LABELS[@]}"; do
  IFS=":" read -r color desc <<< "${LABELS[$name]}"
  gh label create "$name" --color "$color" --description "$desc" --repo "$REPO" 2>/dev/null || true
done

# ---- 2. 建 issue 函数（失败不中断，便于脚本重跑续建）----
make() {
  local title="$1"; local labels="$2"; local body="$3"
  gh issue create --repo "$REPO" --title "$title" --label "$labels" --body "$body" \
    || echo "⚠️ 创建失败（可能已存在）：$title"
}

# ---- 3. M1 ----

  "front,milestone:m1" \
" - [ ] create next-app (App Router, TS, ESLint)，包管理用 pnpm
 - [ ] 接入 TailwindCSS，配置 design tokens
 - [ ] 目录结构：app / components / lib / content
 - [ ] 跑通 pnpm dev"

 - [ ] /zh 与 /en 都能渲染"

  "front,responsive,milestone:m1" \
" - [ ] Header：Logo + 导航 + 语言切换
 - [ ] 移动端汉堡菜单
